\lessontitle{Probability with Counting}

When every outcome is equally likely, count the outcomes with permutations or combinations.

\vspace{0.3em}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|>{\raggedright\arraybackslash}p{6.7in}|}\hline
\rule{0pt}{2.4em}$P(\text{event}) = \dfrac{\text{number of ways the event can happen}}{\text{total number of possible outcomes}}$ \qquad Count both with the same method.\\[1em] \hline
\end{tabular}

\vspace{0.8em}
\textbf{Example 1: choosing a group.} A teacher randomly picks 2 students from 5 girls and 4 boys. What is $P(\text{both girls})$?

\vspace{0.4em}
\begin{tabular}{@{}ll}
Total ways to pick 2 of 9: & ${}_9C_2 = 36$\\
Ways to pick 2 of the 5 girls: & ${}_5C_2 = 10$\\[0.3em]
Probability: & $P(\text{both girls}) = \dfrac{10}{36} = \dfrac{5}{18} \approx 0.28$\\
\end{tabular}

\vspace{0.8em}
\textbf{Example 2: a lottery.} Pick 3 different numbers from 1 to 20. Order does not matter. What is $P(\text{win})$?

\vspace{0.3em}
Total: ${}_{20}C_3 = 1{,}140$. \; Only 1 of them wins. \quad $P(\text{win}) = \dfrac{1}{1{,}140} \approx 0.0009$

\vspace{0.8em}
\textbf{Example 3: jobs.} 10 students are in a club. A president and a vice president are chosen at random.
What is the probability that Ana is president and Ben is vice president?

\vspace{0.3em}
Order matters (different jobs). Total: ${}_{10}P_2 = 90$. \; Only 1 way works. \quad $P = \dfrac{1}{90}$

\vspace{0.8em}
\textbf{Steps}

\vspace{0.2em}
\begin{tabular}{@{}ll}
1. & Decide: does order matter? (permutation or combination)\\
2. & Count all the possible outcomes.\\
3. & Count the outcomes you want.\\
4. & Divide.\\
\end{tabular}

% ---------------- PROBLEMS ----------------
\newpage
\begin{enumerate}

\item A committee of 3 is chosen at random from 6 juniors (11th grade) and 4 seniors (12th grade).
\begin{enumerate}[itemsep=1em]
  \item How many committees are possible? \ans{\quad ${}_{10}C_3 = 120$}
  \item Find $P(\text{all seniors})$. \ans{\quad ${}_4C_3 = 4$. \; $\frac{4}{120} = \frac{1}{30}$}
  \work{1.5em}
  \item Find $P(\text{all juniors})$. \ans{\quad ${}_6C_3 = 20$. \; $\frac{20}{120} = \frac{1}{6}$}
\end{enumerate}
\work{2em}

\item A lottery: pick 4 different numbers from 1 to 30. Order does not matter. What is the probability of winning with one ticket?

\ans{${}_{30}C_4 = 27{,}405$. \; $P(\text{win}) = \frac{1}{27{,}405}$}
\work{3em}

\item 6 runners are in a race. Each order of finish is equally likely. What is the probability that Maya finishes 1st and Leo finishes 2nd?

\ans{Order matters. ${}_6P_2 = 30$. \; $P = \frac{1}{30}$}
\work{3em}

\item Two cards are dealt from a deck of 52. What is $P(\text{both aces})$? (There are 4 aces.)

\ans{${}_4C_2 = 6$ and ${}_{52}C_2 = 1{,}326$. \; $P = \frac{6}{1{,}326} = \frac{1}{221}$}
\work{3em}

\newpage
\item In a taste test, there are 4 drinks from 4 different brands. A person tastes each drink and matches it to a brand. Each brand is used once.
If the person just guesses, what is the probability that all 4 are right?

\ans{There are $4! = 24$ ways to match the brands. Only 1 is right. \; $P = \frac{1}{24}$}
\work{3em}

\item You grab 2 shirts at random from a drawer of 8 shirts. 3 of the shirts are blue. What is $P(\text{both blue})$?

\ans{${}_3C_2 = 3$ and ${}_8C_2 = 28$. \; $P = \frac{3}{28}$}
\work{3em}

\item A big lottery asks you to pick 5 numbers from 1 to 70 and 1 extra number from 1 to 25.
\begin{enumerate}[itemsep=1em]
  \item Use a calculator. How many different tickets are possible?

  \ans{${}_{70}C_5 \cdot 25 = 12{,}103{,}014 \cdot 25 = 302{,}575{,}350$}
  \work{2em}
  \item What is the probability of winning with one ticket? Should you expect to win? Explain.

  \ans{$\frac{1}{302{,}575{,}350}$. No. Winning is extremely unlikely.}
\end{enumerate}

\end{enumerate}
