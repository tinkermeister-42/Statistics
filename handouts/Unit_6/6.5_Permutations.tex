\lessontitle{Permutations}

\textbf{The counting principle.} If one choice has $m$ options and a second choice has $n$ options, there are $m \cdot n$ ways to make both choices.

\vspace{0.2em}
\textbf{Example:} 3 shirts and 4 pairs of pants make $3 \cdot 4 = 12$ outfits.

\vspace{1em}
\textbf{Arranging.} In how many ways can 4 students sit in 4 chairs in a row?

\vspace{0.3em}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|c|c|c|c|}\hline
Chair 1 & Chair 2 & Chair 3 & Chair 4\\ \hline
4 choices & 3 left & 2 left & 1 left\\ \hline
\end{tabular}
\qquad $4 \cdot 3 \cdot 2 \cdot 1 = 24$ ways
\end{center}

\begin{tabular}{|>{\raggedright\arraybackslash}p{6.7in}|}\hline
\textbf{Factorial:} \; $n! = n \cdot (n - 1) \cdot (n - 2) \cdots 2 \cdot 1$ \qquad read ``$n$ factorial''\par\vspace{0.3em}
$4! = 4 \cdot 3 \cdot 2 \cdot 1 = 24$ \qquad $1! = 1$ \qquad $0! = 1$ (there is exactly one way to arrange nothing)\\ \hline
\end{tabular}

\vspace{1em}
\textbf{Permutations.} A \textbf{permutation} is an arrangement where \textbf{order matters}.

\vspace{0.2em}
\textbf{Example:} A club of 10 students picks a president, a vice president, and a secretary. How many ways?

\vspace{0.2em}
$10 \cdot 9 \cdot 8 = 720$ ways. \; (Ana as president and Ben as vice president is different from Ben as president and Ana as vice president.)

\vspace{0.6em}
\begin{tabular}{|>{\raggedright\arraybackslash}p{6.7in}|}\hline
\rule{0pt}{2.4em}\textbf{Permutation formula:} \qquad ${}_nP_r = \dfrac{n!}{(n - r)!}$ \qquad arrange $r$ things chosen from $n$ things\par\vspace{0.6em}
\rule{0pt}{2.2em}${}_{10}P_3 = \dfrac{10!}{7!} = \dfrac{10 \cdot 9 \cdot 8 \cdot \cancel{7 \cdot 6 \cdots 1}}{\cancel{7 \cdot 6 \cdots 1}} = 720$\rule[-1.4em]{0pt}{0pt}\\ \hline
\end{tabular}

\vspace{0.4em}
${}_nP_r$ is read ``$n$ permute $r$.'' Some books write $P(n, r)$.

\vspace{0.8em}
\textbf{Repeats allowed?} 3-letter codes from 26 letters:

\vspace{0.2em}
\begin{tabular}{@{}ll}
No repeats: & $26 \cdot 25 \cdot 24 = 15{,}600$ \quad (a permutation)\\
Repeats allowed: & $26 \cdot 26 \cdot 26 = 26^3 = 17{,}576$ \quad (not a permutation; use the counting principle)\\
\end{tabular}

% ---------------- PROBLEMS ----------------
\newpage
\begin{enumerate}

\item Find each value.
\begin{enumerate}[itemsep=1.4em]
  \item $5!$ \ans{\quad $5 \cdot 4 \cdot 3 \cdot 2 \cdot 1 = 120$}
  \item $\dfrac{6!}{4!}$ \ans{\quad $6 \cdot 5 = 30$}
  \item $0!$ \ans{\quad 1}
  \item ${}_8P_3$ \ans{\quad $8 \cdot 7 \cdot 6 = 336$}
\end{enumerate}
\work{1em}

\item You have 5 shirts, 3 pairs of pants, and 2 pairs of shoes. How many outfits can you make?

\ans{$5 \cdot 3 \cdot 2 = 30$}
\work{2.5em}

\item In how many ways can 6 different books be arranged on a shelf?

\ans{$6! = 720$}
\work{2.5em}

\item 8 runners are in a race. In how many ways can gold, silver, and bronze medals be given?

\ans{${}_8P_3 = 8 \cdot 7 \cdot 6 = 336$}
\work{2.5em}

\newpage
\item A phone code has 4 digits (0 to 9).
\begin{enumerate}[itemsep=1em]
  \item How many codes are there if digits can repeat? \ans{\quad $10 \cdot 10 \cdot 10 \cdot 10 = 10{,}000$}
  \work{1.5em}
  \item How many codes are there if no digit repeats? \ans{\quad ${}_{10}P_4 = 10 \cdot 9 \cdot 8 \cdot 7 = 5{,}040$}
\end{enumerate}
\work{2em}

\item In how many ways can you arrange the letters of the word MATH? \ans{\quad $4! = 24$}
\work{2.5em}

\item A club of 12 members picks a president and a treasurer. One person cannot have both jobs. How many ways?

\ans{${}_{12}P_2 = 12 \cdot 11 = 132$}
\work{2.5em}

\item Explain why ${}_{10}P_3$ is the same as $10 \cdot 9 \cdot 8$.

\ans{$\frac{10!}{7!}$: everything from 7 down to 1 cancels, so only $10 \cdot 9 \cdot 8$ is left.\par\vspace{0.3em}That matches 10 choices, then 9, then 8.}

\end{enumerate}
