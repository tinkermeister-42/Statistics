\lessontitle{Conditional Probability}

A \textbf{conditional probability} is a probability when you already know something happened.

\vspace{0.4em}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|>{\raggedright\arraybackslash}p{6.7in}|}\hline
\rule{0pt}{2.4em}$P(A \mid B) = \dfrac{P(A \cap B)}{P(B)}$ \qquad read ``the probability of $A$ \textbf{given} $B$''\par\vspace{0.3em}
In a table: only look at the $B$ group. \qquad $P(A \mid B) = \dfrac{\text{number in } A \text{ and } B}{\text{number in } B}$\\[0.8em] \hline
\end{tabular}

\vspace{0.6em}
\begin{minipage}[c]{0.4\linewidth}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|l|c|c|c|}\hline
 & \textbf{Job} & \textbf{No job} & \textbf{Total} \\ \hline
\textbf{Sport} & 8 & 22 & 30 \\ \hline
\textbf{No sport} & 12 & 18 & 30 \\ \hline
\textbf{Total} & 20 & 40 & 60 \\ \hline
\end{tabular}
\end{minipage}\hfill
\begin{minipage}[c]{0.56\linewidth}
$P(\text{job} \mid \text{sport}) = \dfrac{8}{30} \approx 0.27$ \quad (only the 30 in sports)

\vspace{0.6em}
$P(\text{sport} \mid \text{job}) = \dfrac{8}{20} = 0.40$ \quad (only the 20 with jobs)

\vspace{0.4em}
The order matters. These are different questions.
\end{minipage}

\vspace{1em}
\textbf{Tree diagrams.} 60\% of students ride the bus. 25\% of bus riders are late. 10\% of the others are late.

\begin{center}\fig{Unit_6/6.3_tree_bus}\end{center}

\begin{tabular}{@{}ll}
Multiply \textbf{along} a path: & $P(\text{bus} \cap \text{late}) = 0.6 \cdot 0.25 = 0.15$\\
Add the paths you want: & $P(\text{late}) = 0.15 + 0.04 = 0.19$\\
Then divide: & $P(\text{bus} \mid \text{late}) = \dfrac{0.15}{0.19} \approx 0.79$ \quad Most late students ride the bus.\\
\end{tabular}

\vspace{0.8em}
\textbf{Independence check.} $A$ and $B$ are independent if $P(A \mid B) = P(A)$.

\vspace{0.2em}
$P(\text{job}) = \dfrac{20}{60} \approx 0.33$ \; but \; $P(\text{job} \mid \text{sport}) \approx 0.27$. \; They are different, so having a job and playing a sport are \textbf{not} independent.

\newpage
\begin{enumerate}

\item 100 students said if they ate breakfast and if they were on time.

\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|l|c|c|c|}\hline
 & \textbf{Breakfast} & \textbf{No breakfast} & \textbf{Total} \\ \hline
\textbf{On time} & 45 & 15 & 60 \\ \hline
\textbf{Late} & 10 & 30 & 40 \\ \hline
\textbf{Total} & 55 & 45 & 100 \\ \hline
\end{tabular}
\end{center}
\begin{enumerate}[itemsep=1.4em]
  \item $P(\text{late})$ \ans{\quad $\frac{40}{100} = 0.40$}
  \item $P(\text{late} \mid \text{no breakfast})$ \ans{\quad $\frac{30}{45} \approx 0.67$}
  \item $P(\text{no breakfast} \mid \text{late})$ \ans{\quad $\frac{30}{40} = 0.75$}
  \item $P(\text{late} \mid \text{breakfast})$ \ans{\quad $\frac{10}{55} \approx 0.18$}
  \item Are being late and eating breakfast independent? Explain.

  \ans{No. $P(\text{late}) = 0.40$, but $P(\text{late} \mid \text{no breakfast}) \approx 0.67$. Knowing about breakfast changes the probability.}
\end{enumerate}
\work{1.5em}

\item $P(A \cap B) = 0.12$ and $P(B) = 0.4$. Find $P(A \mid B)$.

\ans{$\frac{0.12}{0.4} = 0.3$}
\work{2.5em}

\item Draw one card from a deck. Face cards are J, Q, and K (12 in all).
\begin{enumerate}[itemsep=1.4em]
  \item $P(\text{king} \mid \text{face card})$ \ans{\quad $\frac{4}{12} = \frac{1}{3}$}
  \item $P(\text{face card} \mid \text{king})$ \ans{\quad $\frac{4}{4} = 1$. Every king is a face card.}
\end{enumerate}

\newpage
\item A factory has two machines. Machine A makes 70\% of the parts and Machine B makes 30\%.
2\% of A's parts are defective (D). 5\% of B's parts are defective.
\begin{enumerate}[itemsep=1em]
  \item Fill in the tree diagram.

  \begin{center}\tree{A}{0.7}{B}{0.3}{D}{not D}{0.02}{\ans{0.98}}{0.05}{\ans{0.95}}{\ans{$0.7 \cdot 0.02 = 0.014$}}{\ans{$0.7 \cdot 0.98 = 0.686$}}{\ans{$0.3 \cdot 0.05 = 0.015$}}{\ans{$0.3 \cdot 0.95 = 0.285$}}\end{center}
  \item Find $P(D)$, the probability a part is defective. \ans{\quad $0.014 + 0.015 = 0.029$}
  \work{1.5em}
  \item A part is defective. Find $P(B \mid D)$.

  \ans{$\frac{0.015}{0.029} \approx 0.52$. Even though B makes fewer parts, about half of the defective parts come from B.}
\end{enumerate}
\work{2em}

\item 100 students said if they wear glasses and if they play an instrument.
40 play an instrument, and 12 of them wear glasses. 30 of all 100 students wear glasses.
Are wearing glasses and playing an instrument independent? Show your work.

\ans{$P(\text{glasses}) = \frac{30}{100} = 0.3$. \; $P(\text{glasses} \mid \text{instrument}) = \frac{12}{40} = 0.3$. \; They are equal, so yes, independent.}

\end{enumerate}
